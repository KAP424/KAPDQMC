def EEfit_F(x, y, fix, arg):
    if fix == 0:
        A=np.ones((len(x),3))
        A[:,0]=x
        A[:,1]=np.log(x)
        return np.linalg.inv(A.transpose() @A)@A.transpose()@y
    
    elif fix == 1:
        y_adj = y - arg * x
        A=np.ones((len(x),2))
        A[:,0]=np.log(x)
        return np.linalg.inv(A.transpose() @A)@A.transpose()@y_adj
    
    elif fix == 2:
        y_adj = y - arg * np.log(x)
        A=np.ones((len(x),2))
        A[:,0]=x
        return np.linalg.inv(A.transpose() @A)@A.transpose()@y_adj

def powerfit(x,y):
    initial_guess = [0.3, 1.0, 1.0]

    def model_func(x, a, b, c):
        return a + b * x**(c)
    
    params, _ = curve_fit(
        model_func, x, y,
        p0=initial_guess,
        # bounds=None,
        maxfev=100000
    )
    return params

def polyfit(x, y, n):
    A = np.ones((len(x), n+1))
    for i in range(1, n+1):
        A[:, i] = x**i
    coeffs, _, _, _ = np.linalg.lstsq(A, y, rcond=None)
    return coeffs
